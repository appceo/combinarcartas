async function descargarConMultiproxy(url, ms) { // 1. Intento directo try { const c1 = new AbortController(); const t1 = setTimeout(() => c1.abort(), 3000); const r1 = await fetch(url, { signal: c1.signal }); clearTimeout(t1); if (r1.ok) return await r1.arrayBuffer(); } catch (e) {} // 2. Proxy de codificación Base64 via JSON (Bypassea el bloqueo de PDFs de WordPress) try { const c2 = new AbortController(); const t2 = setTimeout(() => c2.abort(), ms); const res = await fetch(`https://api.allorigins.win/get?url=${encodeURIComponent(url)}`, { signal: c2.signal }); clearTimeout(t2); if (res.ok) { const data = await res.json(); if (data.contents) { // Convertir Data URL o texto a ArrayBuffer const base64Data = data.contents.split(',')[1] || data.contents; const binaryString = atob(base64Data); const bytes = new Uint8Array(binaryString.length); for (let i = 0; i < binaryString.length; i++) { bytes[i] = binaryString.charCodeAt(i); } return bytes.buffer; } } } catch (e) {} // 3. Fallback alternativo try { const c3 = new AbortController(); const t3 = setTimeout(() => c3.abort(), ms); const r3 = await fetch(`https://corsproxy.io/?${encodeURIComponent(url)}`, { signal: c3.signal }); clearTimeout(t3); if (r3.ok) return await r3.arrayBuffer(); } catch (e) {} throw new Error("El servidor de origen bloquea la descarga automática del PDF."); }
